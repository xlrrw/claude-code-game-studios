local profiles = {} -- [player] = { data = {}, canSave = bool, saving = bool }
local loadedEvent = Instance.new("BindableEvent")
DataService.Loaded = loadedEvent.Event

local store = nil
do
	local ok, result = pcall(function()
		return DataStoreService:GetDataStore(Config.DataStoreName)
	end)
	if ok then
		store = result
	else
		warn("[Маяк] DataStore недоступен — прогресс не будет сохраняться. Включи: Game Settings → Security → Enable Studio Access to API Services. Ошибка: " .. tostring(result))
	end
end

local function deepCopy(value)
	if type(value) ~= "table" then
		return value
	end
	local copy = {}
	for k, v in pairs(value) do
		copy[k] = deepCopy(v)
	end
	return copy
end

local function reconcile(data, template)
	for key, value in pairs(template) do
		if data[key] == nil then
			data[key] = deepCopy(value)
		elseif type(value) == "table" and type(data[key]) == "table" then
			-- для словарей с фиксированными ключами (upgrades) добиваем недостающие
			if next(value) ~= nil and #value == 0 then
				reconcile(data[key], value)
			end
		end
	end
	return data
end

local function keyFor(player)
	return "player_" .. player.UserId
end

function DataService.Load(player)
	local data = nil
	local loadedOk = false

	if store then
		-- в Studio одна попытка, чтобы не ждать, если доступ к API выключен
		local attempts = RunService:IsStudio() and 1 or 4
		for attempt = 1, attempts do
			local ok, result = pcall(function()
				return store:GetAsync(keyFor(player))
			end)
			if ok then
				data = result
				loadedOk = true
				break
			end
			warn("[Маяк] Не удалось загрузить данные " .. player.Name .. " (попытка " .. attempt .. "): " .. tostring(result))
			if attempt < attempts then
				task.wait(attempt * 1.5)
			end
		end
	end

	if player.Parent == nil then
		return nil
	end

	if type(data) ~= "table" then
		data = deepCopy(TEMPLATE)
		data.firstJoin = os.time()
		if RunService:IsStudio() then
			data.light = Config.Debug.StartingLightInStudio or 0
		end
	end
	reconcile(data, TEMPLATE)

	profiles[player] = {
		data = data,
		-- если загрузка упала, НЕ сохраняем, иначе перезапишем настоящий прогресс пустым
		canSave = store ~= nil and loadedOk,
		saving = false,
	}
	if store and not loadedOk then
		warn("[Маяк] Прогресс " .. player.Name .. " не загрузился — в этой сессии он не будет сохраняться.")
	end

	loadedEvent:Fire(player, data)
	return data
end

function DataService.Get(player)
	local profile = profiles[player]
	return profile and profile.data or nil
end

-- Ждёт загрузки профиля (нужно для обработки покупок).
function DataService.WaitFor(player, timeout)
	local deadline = os.clock() + (timeout or 10)
	while os.clock() < deadline do
		local profile = profiles[player]
		if profile then
			return profile.data
		end
		if player.Parent == nil then
			return nil
		end
		task.wait(0.2)
	end
	return nil
end

function DataService.CanSave(player)
	local profile = profiles[player]
	return profile ~= nil and profile.canSave
end

-- Возвращает true, если сохранение прошло успешно.
function DataService.Save(player)
	local profile = profiles[player]
	if not profile or not profile.canSave then
		return false
	end
	while profile.saving do
		task.wait(0.1)
	end
	profile.saving = true
	local success = false
	for attempt = 1, 3 do
		local ok, err = pcall(function()
			store:UpdateAsync(keyFor(player), function()
				return profile.data
			end)
		end)
		if ok then
			success = true
			break
		end
		warn("[Маяк] Ошибка сохранения " .. player.Name .. " (попытка " .. attempt .. "): " .. tostring(err))
		task.wait(attempt)
	end
	profile.saving = false
	return success
end

function DataService.RecordPurchase(player, purchaseId)
	local data = DataService.Get(player)
	if not data then
		return
	end
	data.purchases[purchaseId] = true
	table.insert(data.purchaseLog, purchaseId)
	while #data.purchaseLog > MAX_PURCHASE_LOG do
		local oldest = table.remove(data.purchaseLog, 1)
		data.purchases[oldest] = nil
	end
end

function DataService.Release(player)
	DataService.Save(player)
	profiles[player] = nil
end

function DataService.Start()
	Players.PlayerRemoving:Connect(function(player)
		DataService.Release(player)
	end)

	task.spawn(function()
		while true do
			task.wait(AUTOSAVE_SECONDS)
			for player in pairs(profiles) do
				task.spawn(DataService.Save, player)
			end
		end
	end)

	game:BindToClose(function()
		if RunService:IsStudio() and not store then
			return
		end
		local pending = 0
		for player in pairs(profiles) do
			pending = pending + 1
			task.spawn(function()
				DataService.Save(player)
				pending = pending - 1
			end)
		end
		local deadline = os.clock() + 25
		while pending > 0 and os.clock() < deadline do

local function makeShardPart(spawnPart, value)
	local part = Instance.new("Part")
	part.Name = "LightShard"
	part.Size = value > Config.Light.ShardValue and Vector3.new(1.3, 1.3, 1.3) or Vector3.new(0.9, 0.9, 0.9)
	part.Material = Enum.Material.Neon
	if value > Config.Light.ShardValue then
		part.Color = Color3.fromRGB(255, 120, 220)
	else
		part.Color = SHARD_COLORS[rng:NextInteger(1, #SHARD_COLORS)]
	end
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.CanQuery = false
	part.CastShadow = false
	part.CFrame = CFrame.new(spawnPart.Position) * CFrame.Angles(math.rad(45), 0, math.rad(45))
	part:SetAttribute("Value", value)

	local attachment = Instance.new("Attachment")
	attachment.Parent = part
	local sparkle = Instance.new("ParticleEmitter")
	sparkle.Texture = "rbxasset://textures/particles/sparkles_main.dds"
	sparkle.Color = ColorSequence.new(part.Color)
	sparkle.LightEmission = 1
	sparkle.Size = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.3),
		NumberSequenceKeypoint.new(1, 0),
	})
	sparkle.Transparency = NumberSequence.new(0.1, 1)
	sparkle.Lifetime = NumberRange.new(0.6, 1)
	sparkle.Rate = 5
	sparkle.Speed = NumberRange.new(0.5, 1.5)
	sparkle.SpreadAngle = Vector2.new(180, 180)
	sparkle.Parent = attachment

	return part
end

local function setupShards()
	activeFolder = world:FindFirstChild("ActiveShards")

local function collect(player, shard)
	shard.active = false
	local position = shard.part.Position
	shard.part.Parent = nil
	local amount = math.max(1, math.floor(shard.value * State.GetMultiplier(player) + 0.5))
	State.AddLight(player, amount)
	State.Remotes.Fx:FireClient(player, "Pickup", position, amount)
	task.delay(Config.Light.ShardRespawnSeconds, function()
		shard.active = true
		shard.part.Parent = activeFolder
	end)
end

local function pickupLoop()
	local accumulator = 0
	RunService.Heartbeat:Connect(function(dt)
		accumulator = accumulator + dt
		if accumulator < Config.Light.PickupCheckInterval then
			return
		end
		accumulator = 0
		for _, player in ipairs(Players:GetPlayers()) do
			local character = player.Character
			local root = character and character:FindFirstChild("HumanoidRootPart")
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
			if root and humanoid and humanoid.Health > 0 and DataService.Get(player) then
				local radius = State.GetRadius(player)
				local position = root.Position
				for _, shard in ipairs(shards) do
					if shard.active and (shard.part.Position - position).Magnitude <= radius then
						collect(player, shard)
					end
				end
			end
		end
	end)
end

local function recomputeLevel(announce)
	local thresholds = Config.Lighthouse.LevelThresholds
	local w = State.World
	local oldLevel = w.level
	local level = 1
	for i, threshold in ipairs(thresholds) do
		if w.lighthouseTotal >= threshold then
			level = i
		end
	end
	w.level = level
	w.prevThreshold = thresholds[level]
	w.nextThreshold = thresholds[level + 1]
	applyLighthouseVisuals()
	if level ~= oldLevel then
		State.PushAllPlayers()
		if announce and level > oldLevel then
			State.NotifyAll("🗼 Маяк разгорелся до уровня " .. level .. "! Весь сервер получает +" .. math.floor((level - 1) * Config.Lighthouse.BonusPerLevel * 100 + 0.5) .. "% света", "level")
			State.Remotes.Fx:FireAllClients("LighthouseLevel", level)
		end
	end
	State.PushWorld()
end

function LightService.Deposit(player, fraction)
	if type(fraction) ~= "number" then
		return
	end
	if fraction ~= 0.25 and fraction ~= 0.5 and fraction ~= 1 then
		return
	end
	local data = DataService.Get(player)
	if not data then
		return
	end
	local amount = math.floor(data.light * fraction)
	if amount < 1 then
		State.Notify(player, "У тебя пока нет света. Собирай сияющие осколки!", "warn")
		return
	end
	if not State.SpendLight(player, amount) then
		return
	end
	State.AddXp(player, amount * Config.Lighthouse.XpPerLight)
	State.World.lighthouseTotal = State.World.lighthouseTotal + amount

	local shadow = State.World.shadow
	if shadow.active then
		shadow.progress = shadow.progress + amount
		contributors[player] = true
	end

	State.Remotes.Fx:FireClient(player, "Deposit", amount)
	State.Notify(player, "Ты отдал Маяку " .. State.FormatNumber(amount) .. " ✦ и получил столько же опыта Смотрителя", "deposit")
	recomputeLevel(true)
end

local function runShadowEvent()
	local cfg = Config.ShadowEvent
	local shadow = State.World.shadow
	contributors = {}
	shadow.active = true
	shadow.progress = 0
	shadow.goal = math.max(cfg.GoalMin, #Players:GetPlayers() * cfg.GoalPerPlayer)
	shadow.endsAt = State.Now() + cfg.Duration
	State.NotifyAll("🌑 ТЕНЬ НАСТУПАЕТ! Вложите в Маяк " .. shadow.goal .. " ✦ за " .. math.floor(cfg.Duration / 60) .. " минуты!", "shadow")
	State.Remotes.Fx:FireAllClients("ShadowStart")
	State.PushWorld()

	while State.Now() < shadow.endsAt and shadow.progress < shadow.goal do
		task.wait(0.5)
		State.PushWorld()
	end

	shadow.active = false
	if shadow.progress >= shadow.goal then
		State.World.shadowBoostUntil = State.Now() + cfg.SuccessBoostMinutes * 60
		State.NotifyAll("☀️ Тень отступила! Весь сервер получает +" .. math.floor(cfg.SuccessBonus * 100) .. "% света на " .. cfg.SuccessBoostMinutes .. " минут!", "victory")
		for player in pairs(contributors) do
			if player.Parent then
				State.AddLight(player, cfg.RewardLight)
				State.AddXp(player, cfg.RewardLight)
				State.Notify(player, "Награда героя: +" .. cfg.RewardLight .. " ✦", "purchase")
			end
		end
		State.Remotes.Fx:FireAllClients("ShadowEnd", true)
		task.delay(cfg.SuccessBoostMinutes * 60 + 1, function()
			State.PushAllPlayers()
			State.PushWorld()
		end)
	else
		State.World.lighthouseTotal = math.floor(State.World.lighthouseTotal * (1 - cfg.FailPenalty))
		State.NotifyAll("💨 Тень погасила часть огня... Маяк потускнел. В следующий раз — вместе!", "shadow")
		State.Remotes.Fx:FireAllClients("ShadowEnd", false)
	end
	contributors = {}
	State.PushAllPlayers()
	recomputeLevel(false)
end

local function shadowLoop()
	local cfg = Config.ShadowEvent
	task.spawn(function()
		local delaySeconds = cfg.FirstDelaySeconds
		while true do
			State.World.shadow.nextAt = State.Now() + delaySeconds
			State.PushWorld()
			task.wait(delaySeconds)
			if #Players:GetPlayers() > 0 then
				runShadowEvent()
			end
			delaySeconds = rng:NextInteger(cfg.IntervalMin, cfg.IntervalMax)
		end
	end)
end

local function processReceipt(receiptInfo)
	local player = Players:GetPlayerByUserId(receiptInfo.PlayerId)
	if not player then
		-- игрок вышел — Roblox повторит, когда он зайдёт снова
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local data = DataService.WaitFor(player, 15)
	if not data then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	if data.purchases[receiptInfo.PurchaseId] then
		return Enum.ProductPurchaseDecision.PurchaseGranted
	end

	local key = productKeyById[receiptInfo.ProductId]
	local handler = key and handlers[key]
	if not handler then
		warn("[Маяк] Неизвестный товар " .. tostring(receiptInfo.ProductId) .. " — проверь MonetizationIds")
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local ok, err = pcall(handler, player, data)
	if not ok then
		warn("[Маяк] Ошибка выдачи товара " .. key .. ": " .. tostring(err))
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	DataService.RecordPurchase(player, receiptInfo.PurchaseId)

	if DataService.CanSave(player) then
		if not DataService.Save(player) then
			-- запись о покупке осталась в памяти: при повторе Roblox товар не выдастся дважды
			return Enum.ProductPurchaseDecision.NotProcessedYet
		end
	end
	return Enum.ProductPurchaseDecision.PurchaseGranted
end

local function setupCheckpoints()
	local folder = obby and obby:FindFirstChild("Checkpoints")
	if not folder then
		return
	end
	for _, cp in ipairs(folder:GetChildren()) do
		if cp:IsA("BasePart") then
			cp.Touched:Connect(function(hit)
				local player = playerFromHit(hit)
				if not player then
					return
				end
				local current = checkpoints[player]
				local index = cp:GetAttribute("Index") or 0
				if current == cp then
					return
				end
				if current and (current:GetAttribute("Index") or 0) >= index then
					return
				end
				checkpoints[player] = cp
				State.Remotes.Fx:FireClient(player, "Checkpoint", cp.Position, index)
			end)
		end
	end
end

local function setupFalling()
	local folder = obby and obby:FindFirstChild("FallingPlatforms")
	if not folder then
		return
	end
	for _, platform in ipairs(folder:GetChildren()) do
		if platform:IsA("BasePart") then
			local origin = platform.CFrame
			local originColor = platform.Color
			local originTransparency = platform.Transparency
			-- светящийся обод (дочерняя деталь Rim) двигаем вместе с платформой
			local rim = platform:FindFirstChild("Rim")
			local rimOffset = rim and origin:ToObjectSpace(rim.CFrame)
			local rimTransparency = rim and rim.Transparency
			local function moveRim(cf, transparency, time)
				if not rim then
					return
				end
				if time then
					TweenService:Create(rim, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
						CFrame = cf * rimOffset,
						Transparency = transparency,
					}):Play()
				else
					rim.CFrame = cf * rimOffset
				end
			end
			local busy = false
			platform.Touched:Connect(function(hit)
				if busy or not playerFromHit(hit) then
					return
				end
				busy = true
				platform.Color = Color3.fromRGB(255, 110, 140)
				-- дрожь
				local shakes = math.max(2, math.floor(Config.Obby.FallWarnSeconds / 0.08))
				for i = 1, shakes do
					local shake = origin * CFrame.new((i % 2 == 0 and 0.12 or -0.12), 0, 0)
					platform.CFrame = shake
					moveRim(shake)
					task.wait(0.08)
				end
				platform.CanCollide = false
				local fallen = origin * CFrame.new(0, -25, 0) * CFrame.Angles(math.rad(25), 0, math.rad(15))
				TweenService:Create(platform, TweenInfo.new(0.9, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
					CFrame = fallen,
					Transparency = 1,
				}):Play()
				moveRim(fallen, 1, 0.9)
				task.wait(Config.Obby.FallRespawnSeconds)
				platform.CFrame = origin * CFrame.new(0, -3, 0)
				moveRim(origin * CFrame.new(0, -3, 0))
				platform.Color = originColor
				platform.CanCollide = true
				TweenService:Create(platform, TweenInfo.new(0.4, Enum.EasingStyle.Back), {
					CFrame = origin,
					Transparency = originTransparency,
				}):Play()
				if rim then
					TweenService:Create(rim, TweenInfo.new(0.4, Enum.EasingStyle.Back), {
						CFrame = origin * rimOffset,
						Transparency = rimTransparency,
					}):Play()
				end
				task.wait(0.4)
				busy = false
			end)
		end
	end
end

local function setupBlink()
	local folder = obby and obby:FindFirstChild("BlinkPlatforms")
	if not folder then
		return
	end
	local groups = { [0] = {}, [1] = {} }
	for _, platform in ipairs(folder:GetChildren()) do
		if platform:IsA("BasePart") then
			local phase = platform:GetAttribute("Phase") or 0
			table.insert(groups[phase % 2], platform)
		end
	end
	task.spawn(function()
		local phase = 0
		while true do
			for groupPhase, list in pairs(groups) do
				local solid = groupPhase == phase
				for _, platform in ipairs(list) do
					platform.CanCollide = solid
					platform.Transparency = solid and 0 or 0.8
					platform.Material = solid and Enum.Material.Neon or Enum.Material.ForceField
				end
			end
			-- предупреждение: за 0.5 сек до смены твёрдые платформы мерцают
			task.wait(Config.Obby.BlinkPeriodSeconds - 0.5)
			for _, platform in ipairs(groups[phase]) do
				platform.Transparency = 0.4
			end
			task.wait(0.5)
			phase = 1 - phase
		end
	end)
end

local function setupFinish()
	local finish = obby and obby:FindFirstChild("Finish", true)
	if not finish or not finish:IsA("BasePart") then
		return
	end
	finish.Touched:Connect(function(hit)
		local player = playerFromHit(hit)
		if not player then
			return
		end
		local now = os.clock()
		if lastFinish[player] and now - lastFinish[player] < Config.Obby.FinishCooldownSeconds then
			return
		end
		lastFinish[player] = now
		local data = DataService.Get(player)
		if not data then
			return
		end
		data.obbyFinishes = data.obbyFinishes + 1
		local reward = math.floor(Config.Obby.FinishReward * State.GetMultiplier(player) + 0.5)
		State.AddLight(player, reward)
		State.AddXp(player, Config.Obby.FinishXp)
		State.Remotes.Fx:FireClient(player, "ObbyFinish", reward)
		if data.starShards < 1 then
			data.starShards = 1
			State.Remotes.Dialogue:FireClient(player, {
				speaker = "Осколок Звезды",
				lines = { Config.Story.StarShardMessage },
			})
		else
			State.Notify(player, "Паркур пройден! +" .. State.FormatNumber(reward) .. " ✦", "victory")
		end
		State.PushPlayer(player)
		task.delay(3, function()
			if player.Parent then
				ObbyService.ReturnToHub(player)
			end
		end)
	end)
end

local function setupPortals()
	-- любые детали с атрибутом Teleport = "Hub" возвращают на остров с Маяком
	for _, inst in ipairs(world:GetDescendants()) do
		if inst:IsA("BasePart") and inst:GetAttribute("Teleport") == "Hub" then
			local debounce = {}
			inst.Touched:Connect(function(hit)
				local player = playerFromHit(hit)
				if player and not debounce[player] then
					debounce[player] = true
					ObbyService.ReturnToHub(player)
					task.delay(1, function()
						debounce[player] = nil
					end)
				end
			end)
		end
	end
end

local function voidLoop()
	task.spawn(function()
		while true do
			task.wait(0.25)
			for _, player in ipairs(Players:GetPlayers()) do
				local character = player.Character
				local root = character and character:FindFirstChild("HumanoidRootPart")
				if root and root.Position.Y < Config.Obby.VoidY then
					local cp = checkpoints[player]
					if cp and cp.Parent then
						teleport(player, cp.CFrame + Vector3.new(0, 4, 0))
					else
						teleport(player, hubSpawnCFrame())
					end
					State.Remotes.Fx:FireClient(player, "Respawned")
				end
			end
		end
	end)
end

	setupFalling()
	setupBlink()
	setupFinish()
	setupPortals()
	voidLoop()
	Players.PlayerRemoving:Connect(function(player)
		checkpoints[player] = nil
		lastFinish[player] = nil
	end)
end

	local nextRank = Config.Ranks[index + 1]
	return index, Config.Ranks[index], nextRank
end

function State.GetMultiplier(player)
	local data = DataService.Get(player)
	local passes = State.Passes[player] or {}
	local now = State.Now()
	local mult = 1

	if passes.VIP then
		mult = mult + Config.Passes.VIP.lightBonus
	end
	if player.MembershipType == Enum.MembershipType.Premium then
		mult = mult + Config.Premium.LightBonus
	end
	mult = mult + (State.World.level - 1) * Config.Lighthouse.BonusPerLevel
	if State.World.globalBoostUntil > now then
		mult = mult + Config.Products.GlobalBoost.bonus
	end
	if State.World.shadowBoostUntil > now then
		mult = mult + Config.ShadowEvent.SuccessBonus
	end
	if data then
		local rankIndex = State.GetRank(data.keeperXp)
		mult = mult + (rankIndex - 1) * Config.RankBonus
		for _, petId in ipairs(data.equipped) do
			local pet = Config.PetById[petId]
			if pet then
				mult = mult + pet.lightBonus
			end
		end
	end
	return mult
end

function State.GetRadius(player)
	local data = DataService.Get(player)
	local radius = Config.Light.BasePickupRadius
	if data then
		radius = radius + (data.upgrades.Radius or 0) * Config.Upgrades.Radius.perLevel
		for _, petId in ipairs(data.equipped) do
			local pet = Config.PetById[petId]
			if pet then
				radius = radius + pet.radiusBonus
			end
		end
	end
	return radius
end

function State.GetWalkSpeed(player)
	local data = DataService.Get(player)
	local level = data and data.upgrades.Speed or 0
	return 16 + level * Config.Upgrades.Speed.perLevel
end

function State.GetSlots(player)
	local passes = State.Passes[player] or {}
	if passes.PetSlots then
		return Config.PassEquipSlots
	end
	return Config.BaseEquipSlots
end

local function formatNumber(n)
	n = math.floor(n)
	if n >= 1000000 then
		return string.format("%.1fM", n / 1000000)
	elseif n >= 10000 then
		return string.format("%.1fK", n / 1000)
	end
	return tostring(n)
end

function ShopService.EquipPet(player, petId)

	local data = DataService.Get(player)
	if not def or not data or not data.pets[petId] then
		return
	end

			table.remove(data.equipped, index)
			PetService.Refresh(player)
			State.PushPlayer(player)
			return
		end
	end

			State.Notify(player, "Все слоты заняты. Убери одного духа или купи «Ещё два духа»", "warn")
			return
		end
	else

	end
	PetService.Refresh(player)
	State.PushPlayer(player)
end

		end
	end

local function animateShards(t)

	if not folder then
		return
	end
	for _, part in ipairs(folder:GetChildren()) do
		local base = shardBase[part]
		if not base and part:IsA("BasePart") then

			base = shardBase[part]

		if base then

local function setupLighthouse()
	local lighthouse = world:FindFirstChild("Lighthouse")
	if not lighthouse then
		return
	end
	beamPivot = lighthouse:FindFirstChild("BeamPivot", true)
	if beamPivot then
		beamPivotBase = beamPivot.CFrame
		for _, child in ipairs(beamPivot:GetDescendants()) do
			if child:IsA("Beam") then
				table.insert(beams, { beam = child, width = child.Width1 })
			end
		end
	end
	local function applyLevel()
		local level = lighthouse:GetAttribute("Level") or 1
		local k = 0.6 + level * 0.12
		for _, b in ipairs(beams) do
			b.beam.Width1 = b.width * k
		end
	end
	lighthouse:GetAttributeChangedSignal("Level"):Connect(applyLevel)
	applyLevel()
end

	if beamPivot and beamPivotBase then

		return

			tween(cc, 3, { TintColor = lighting.colorBase.TintColor, Saturation = lighting.colorBase.Saturation, Brightness = lighting.colorBase.Brightness })

		end
	end
	if active then
		Sound.Play("Shadow")
	end
end

	local world = workspace:WaitForChild("LastLighthouseWorld")
	local obby = world:FindFirstChild("Obby")
	local pads = obby and obby:FindFirstChild("JumpPads")
	if not pads then
		return
	end
	for _, pad in ipairs(pads:GetChildren()) do
		if pad:IsA("BasePart") then
			pad.Touched:Connect(function(hit)
				local character = player.Character
				if not character or not hit:IsDescendantOf(character) then
					return
				end
				if os.clock() - lastPad < 0.6 then
					return
				end
				lastPad = os.clock()
				local root = character:FindFirstChild("HumanoidRootPart")
				local humanoid = character:FindFirstChildOfClass("Humanoid")
				if root and humanoid then
					local v = root.AssemblyLinearVelocity
					root.AssemblyLinearVelocity = Vector3.new(v.X, Config.Obby.JumpPadVelocity, v.Z)
					humanoid:ChangeState(Enum.HumanoidStateType.Freefall)
					usedDoubleJump = false
					if effects then
						effects.Burst(pad.Position + Vector3.new(0, 1, 0), Color3.fromRGB(110, 240, 160), 30, 0.7)
						effects.Ring(pad.Position + Vector3.new(0, 0.5, 0), Color3.fromRGB(110, 240, 160))
					end
				end
			end)
		end
	end
end

function Movement.Init(effectsModule)
	effects = effectsModule
	if player.Character then
		task.spawn(onCharacter, player.Character)
	end
	player.CharacterAdded:Connect(onCharacter)

	UserInputService.JumpRequest:Connect(function()
		if not player:GetAttribute("HasWings") or usedDoubleJump then
			return
		end
		local character = player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root then
			return
		end
		if humanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
			return
		end
		usedDoubleJump = true
		local v = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(v.X, 55, v.Z)
		humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		wingBurst(root)
	end)

	task.spawn(setupJumpPads)
end

local function new(className, props, children)
	local inst = Instance.new(className)
	local parent = nil
	if props then
		for k, v in pairs(props) do
			if k == "Parent" then
				parent = v
			else
				inst[k] = v
			end
		end
	end
	if children then
		for _, child in ipairs(children) do
			child.Parent = inst
		end
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function formatNumber(n)
	n = math.floor(n or 0)
	if n >= 1000000 then
		return string.format("%.2fM", n / 1000000)
	elseif n >= 10000 then
		return string.format("%.1fK", n / 1000)
	end
	local s = tostring(n)
	local formatted = s:reverse():gsub("(%d%d%d)", "%1 "):reverse()
	return (formatted:gsub("^ ", ""))
end
UI.FormatNumber = formatNumber

local function formatTime(seconds)
	seconds = math.max(0, math.floor(seconds))
	return string.format("%d:%02d", math.floor(seconds / 60), seconds % 60)
end

		UI.CloseWindow(openWindow, true)

	obj.scale.Scale = 0.75
	tween(obj.overlay, 0.2, { BackgroundTransparency = 0.45 })

function UI.CloseWindow(obj, instant)
	if openWindow == obj then
		openWindow = nil
	end
	if instant then
		obj.overlay.Visible = false
		return
	end
	tween(obj.overlay, 0.15, { BackgroundTransparency = 1 })
	local t = tween(obj.scale, 0.15, { Scale = 0.8 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
	t.Completed:Connect(function()
		if openWindow ~= obj then
			obj.overlay.Visible = false
		end
	end)

local function renderUpgrades(list)
	for order, id in ipairs(Config.UpgradeOrder) do
		local def = Config.Upgrades[id]
		local level = ps and ps.upgrades[id] or 0
		local maxed = level >= def.maxLevel
		local cost = Config.GetUpgradeCost(id, level)
		local c = card(list, order, C.Cyan)
		cardIcon(c, def.icon, C.Cyan)
		local effect
		if id == "Radius" then
			effect = "Сейчас: радиус " .. string.format("%.1f", ps and ps.radius or Config.Light.BasePickupRadius)
		else
			effect = "Сейчас: скорость " .. string.format("%.1f", 16 + level * def.perLevel)
		end
		cardText(c, def.name .. "  ·  " .. level .. "/" .. def.maxLevel, C.Text, def.description, effect)
		if maxed then
			cardButton(c, "МАКС ✓", C.Green, C.Cyan, C.Night, function() end)
		else
			local canAfford = ps and ps.light >= cost
			local b = cardButton(c, formatNumber(cost) .. " ✦", C.Gold, C.Gold2, C.Night, function()
				send("BuyUpgrade", id)
			end)
			if not canAfford then
				setButtonColors(b, Color3.fromRGB(110, 100, 150), Color3.fromRGB(70, 60, 110))
				b.TextColor3 = C.Text
			end
		end
	end
end

local function isEquipped(petId)

	return false
end

local function playDialogue(entry)
	local style = DIALOGUE_STYLES[entry.style or "default"] or DIALOGUE_STYLES.default
	dialogue.gradient.Color = ColorSequence.new(style.c1, style.c2)
	dialogue.stroke.Color = style.accent
	dialogue.nameTagGradient.Color = ColorSequence.new(style.name, style.accent)
	dialogue.name.Text = entry.speaker or ""
	local box = dialogue.box
	box.Visible = true
	dialogue.scale.Scale = 0.85
	tween(dialogue.scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)

	for index, line in ipairs(entry.lines or {}) do
		dialogue.text.Text = line
		dialogue.text.MaxVisibleGraphemes = 0
		dialogue.hint.Text = index < #entry.lines and "Нажми, чтобы продолжить ▶" or "Нажми, чтобы закрыть ✓"
		local skipped = false
		local conn = dialogue.advance.Event:Connect(function()
			skipped = true
		end)
		local length = utf8.len(line) or #line
		local shown = 0
		while shown < length and not skipped do
			shown = shown + 1
			dialogue.text.MaxVisibleGraphemes = shown
			task.wait(0.022)
		end
		conn:Disconnect()
		dialogue.text.MaxVisibleGraphemes = -1
		dialogue.advance.Event:Wait()
	end
	local t = tween(dialogue.scale, 0.15, { Scale = 0.85 })
	t.Completed:Wait()
	box.Visible = false
end

-- Показывает диалог (в очереди). Если передан done (BindableEvent) — сработает по окончании.
function UI.ShowDialogue(entry, done)
	table.insert(dialogue.queue, { entry = entry, done = done })
	if dialogue.running then
		return
	end
	dialogue.running = true
	task.spawn(function()
		while #dialogue.queue > 0 do
			local item = table.remove(dialogue.queue, 1)
			playDialogue(item.entry)
			if item.done then
				item.done:Fire()
			end
		end
		dialogue.running = false
	end)
end

function UI.ShowDialogueAsync(entry)
	local done = Instance.new("BindableEvent")
	UI.ShowDialogue(entry, done)
	done.Event:Wait()
	done:Destroy()
end

			return a.LayoutOrder < b.LayoutOrder
		end)

	end

local function resolve(path)
	local node = game
	for part in string.gmatch(path, "[^%.]+") do
		if node == game then
			node = game:GetService(part)
		else
			node = node:FindFirstChild(part) or node:WaitForChild(part, 5)
		end
	end
	return node
end

function B.island(center, radius)
	terrain:FillCylinder(CFrame.new(center - Vector3.new(0, 2, 0)), 4, radius, Enum.Material.Grass)
	terrain:FillCylinder(CFrame.new(center - Vector3.new(0, 6, 0)), 4, radius * 0.97, Enum.Material.Ground)
	local layers = 7
	local depthStep = math.max(4, radius * 0.16)
	for i = 1, layers do
		local f = i / (layers + 1)
		local r = radius * (1 - f) * 0.95
		local y = center.Y - 8 - i * depthStep
		terrain:FillCylinder(CFrame.new(center.X, y, center.Z), depthStep + 2, r, i % 2 == 0 and Enum.Material.Rock or Enum.Material.Slate)
		local below = center.Y - y
		for _ = 1, 3 do
			local a = B.rng:NextNumber(0, math.pi * 2)
			local br = math.min(r * 0.35, below - 6)
			if br > 2 then
				terrain:FillBall(Vector3.new(center.X + math.cos(a) * r * 0.75, y, center.Z + math.sin(a) * r * 0.75), br, Enum.Material.Rock)
			end
		end
	end
	-- висящие кристаллы под островом
	for _ = 1, math.floor(radius / 6) do
		local a = B.rng:NextNumber(0, math.pi * 2)
		local d = B.rng:NextNumber(0, radius * 0.45)
		local length = B.rng:NextNumber(4, 10)
		local y = center.Y - 10 - (1 - d / radius) * radius * 0.9
		local crystal = B.part(decor, {
			Name = "HangingCrystal",
			Size = Vector3.new(1.2, length, 1.2),
			CFrame = CFrame.new(center.X + math.cos(a) * d, y - length / 2, center.Z + math.sin(a) * d) * CFrame.Angles(0, a, math.rad(B.rng:NextNumber(-10, 10))),
			Color = B.rng:NextNumber() > 0.5 and B.palette.cyan or B.palette.pink,
			Material = Enum.Material.Neon,
			CanCollide = false,
			CastShadow = false,
		})
		crystal.Transparency = 0.1
	end
end

-- Каменный «мостик» из парящих ступенек между двумя точками.
function B.stoneBridge(from, to, color)
	local delta = to - from
	local flat = Vector3.new(delta.X, 0, delta.Z)
	local steps = math.max(math.ceil(flat.Magnitude / 10), math.ceil(math.abs(delta.Y) / 3.5))
	for i = 1, steps - 1 do
		local p = from + delta * (i / steps)
		B.part(decor, {
			Name = "SkyStone",
			Size = Vector3.new(6, 1.5, 6),
			CFrame = CFrame.new(p - Vector3.new(0, 0.75, 0)) * CFrame.Angles(0, B.rng:NextNumber(0, 6.28), 0),
			Color = color or B.palette.stone,
			Material = Enum.Material.Slate,
		})
		local glow = B.part(decor, {
			Name = "SkyStoneGlow",
			Size = Vector3.new(4, 0.3, 4),
			Position = p - Vector3.new(0, 1.7, 0),
			Color = B.palette.cyan,
			Material = Enum.Material.Neon,
			CanCollide = false,
			CastShadow = false,
		})
		glow.Transparency = 0.3
		if i % 2 == 0 then
			B.shardSpawn(p + Vector3.new(0, 2.5, 0))
		end
	end
end

function B.isFree(p)
	if (p - LIGHTHOUSE_POS).Magnitude < 20 then
		return false
	end
	if math.abs(p.X) < 9 and p.Z > -20 and p.Z < 52 then
		return false
	end
	if p.X > 10 and math.abs(p.Z) < 9 then
		return false
	end
	if (p - Vector3.new(16, 0, -6)).Magnitude < 7 then
		return false
	end
	return true
end

-- Высота земли (ландшафта) под точкой.
function B.ground(p)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Include
	params.FilterDescendantsInstances = { terrain }
	local ok, result = pcall(function()
		return workspace:Raycast(p + Vector3.new(0, 40, 0), Vector3.new(0, -80, 0), params)
	end)
	if ok and result then
		return Vector3.new(p.X, result.Position.Y, p.Z)
	end
	return p
end

function B.randomSpot(minR, maxR, center)
	center = center or Vector3.new()
	for _ = 1, 40 do
		local a = B.rng:NextNumber(0, math.pi * 2)
		local d = B.rng:NextNumber(minR, maxR)
		local p = center + Vector3.new(math.cos(a) * d, 0, math.sin(a) * d)
		if center.Magnitude > 1 or B.isFree(p) then
			return B.ground(p)
		end
	end
	return B.ground(center + Vector3.new(minR, 0, 0))
end

		B.part(decor, { Name = "Rope", Size = Vector3.new(25, 0.25, 0.25), Position = Vector3.new(85.5, 2.5, z), Color = B.palette.gold, Material = Enum.Material.Neon, CastShadow = false })
		for _, x in ipairs({ 74, 97 }) do
			B.part(decor, { Name = "BridgePost", Size = Vector3.new(0.6, 3.5, 0.6), Position = Vector3.new(x, 1.5, z), Color = B.palette.wood, Material = Enum.Material.Wood })
		end
	end
	local sign = B.part(decor, { Name = "Sign", Size = Vector3.new(0.4, 3, 8), Position = Vector3.new(70, 4.5, 6), Color = B.palette.wood, Material = Enum.Material.Wood })
	B.billboard(sign, "⚠️ Рушащийся остров →", Color3.fromRGB(255, 180, 140), 3, 280, 80)

	local obby = B.folder(world, "Obby")
	local folders = {
		S = B.folder(obby, "Static"),
		F = B.folder(obby, "FallingPlatforms"),
		B = B.folder(obby, "BlinkPlatforms"),
		J = B.folder(obby, "JumpPads"),
		C = B.folder(obby, "Checkpoints"),
	}
	local colors = {
		S = { body = Color3.fromRGB(120, 100, 210), rim = B.palette.cyan },
		F = { body = Color3.fromRGB(230, 150, 120), rim = Color3.fromRGB(255, 120, 90) },
		C = { body = Color3.fromRGB(80, 70, 140), rim = B.palette.green },
	}

	local function platform(kind, top, size, phase)
		local thickness = 1.4
		local center = top - Vector3.new(0, thickness / 2, 0)
		if kind == "B" then
			local p = B.part(folders.B, {
				Name = "Blink",
				Size = Vector3.new(size, thickness, size),
				Position = center,
				Color = phase == 0 and B.palette.cyan or B.palette.pink,
				Material = Enum.Material.Neon,
				CastShadow = false,
			})
			p:SetAttribute("Phase", phase)
			return p
		end
		local folder = kind == "F" and folders.F or folders.S
		local sx, sz = size, size
		if type(size) == "table" then
			sx, sz = size[1], size[2]
		end
		local c = colors[kind] or colors.S
		local p = B.part(folder, {
			Name = kind == "F" and "Falling" or "Platform",
			Size = Vector3.new(sx, thickness, sz),
			Position = center,
			Color = c.body,
			Material = kind == "F" and Enum.Material.Slate or Enum.Material.SmoothPlastic,
		})
		local rim = B.part(kind == "F" and p or folders.S, {
			Name = "Rim",
			Size = Vector3.new(sx + 0.5, 0.35, sz + 0.5),
			Position = center - Vector3.new(0, thickness / 2 + 0.05, 0),
			Color = c.rim,
			Material = Enum.Material.Neon,
			CanCollide = false,
			CastShadow = false,
		})
		return p
	end

	local cpIndex = 0
	local function checkpoint(top)
		platform("C", top, 14)
		cpIndex = cpIndex + 1
		local cp = B.cyl(folders.C, "Checkpoint" .. cpIndex, top + Vector3.new(0, 0.2, 0), 0.4, 7, B.palette.green, Enum.Material.Neon, { CastShadow = false })
		cp.Transparency = 0.2
		cp:SetAttribute("Index", cpIndex)
		B.part(folders.S, { Name = "FlagPole", Size = Vector3.new(0.3, 7, 0.3), Position = top + Vector3.new(5, 3.5, 5), Color = B.palette.metal, Material = Enum.Material.Metal })
		local flag = B.part(folders.S, { Name = "Flag", Size = Vector3.new(0.1, 1.8, 2.8), Position = top + Vector3.new(5, 6, 3.5), Color = B.palette.green, Material = Enum.Material.Neon, CanCollide = false })
		B.billboard(flag, "Чекпоинт " .. cpIndex, B.palette.green, 2, 160, 60)
	end

	local cursor = Vector3.new(104, 0, 0)
	checkpoint(cursor)
	local path = {
		-- разминка
		{ 12, 1, 0, "S", 8 }, { 12, 1, 4, "S", 7 }, { 12, 1, -4, "S", 7 }, { 12, 0, 0, "S", 8 },
		-- падающие платформы
		{ 12, 0, 3, "F", 7 }, { 12, 1, -3, "F", 7 }, { 12, 0, 3, "F", 7 }, { 12, 1, -3, "F", 7 }, { 12, 0, 0, "F", 7 },
		{ 14, 0, 0, "CP" },
		-- мигающие платформы с островками отдыха
		{ 12, 0, 0, "B0", 7 }, { 11, 0, 0, "S", 6 }, { 11, 1, 0, "B1", 7 }, { 11, 0, 0, "S", 6 }, { 11, 1, 0, "B0", 7 }, { 11, 0, 0, "S", 6 },
		{ 13, 0, 0, "CP" },
		-- батут
		{ 13, 0, 0, "J", 9 }, { 16, 14, 0, "S", 10 }, { 12, 1, 0, "S", 7 },
		-- узкие балки зигзагом
		{ 13, 0, 3.5, "S", { 12, 3 } }, { 13, 0, -7, "S", { 12, 3 } }, { 13, 0, 7, "S", { 12, 3 } }, { 13, 0, -3.5, "S", { 12, 3 } },
		{ 14, 0, 0, "CP" },
		-- финальное испытание
		{ 12, 1, 0, "F", 6 }, { 11, 0, 3, "B1", 6 }, { 11, 0, 0, "S", 5 }, { 11, 1, -3, "F", 6 }, { 11, 0, 0, "B0", 6 }, { 11, 0, 0, "S", 5 }, { 11, 1, 0, "F", 6 },
		-- лестница к вершине
		{ 12, 3, 0, "S", 7 }, { 11, 3, 4, "S", 7 }, { 11, 3, -4, "S", 7 }, { 11, 3, 0, "S", 7 },
	}
	for i, step in ipairs(path) do
		cursor = cursor + Vector3.new(step[1], step[2], step[3])
		local kind = step[4]
		if kind == "CP" then
			checkpoint(cursor)
		elseif kind == "J" then
			platform("S", cursor, step[5])
			local pad = B.cyl(folders.J, "JumpPad", cursor + Vector3.new(0, 0.25, 0), 0.5, 5.5, B.palette.green, Enum.Material.Neon, { CastShadow = false })
			B.sparkles(pad, B.palette.green, 10, 0.4)
			B.billboard(pad, "⬆ БАТУТ", B.palette.green, 2.5, 140, 50)
		elseif kind == "B0" or kind == "B1" then
			platform("B", cursor, step[5], kind == "B0" and 0 or 1)
		else
			platform(kind, cursor, step[5])
		end
		if i % 3 == 0 and kind ~= "CP" then
			B.shardSpawn(cursor + Vector3.new(0, 3, 0), 12)
		end
	end

	-- вершина: остров с Осколком Звезды
	local summit = cursor + Vector3.new(30, 0, 0)
	B.island(summit, 24)
	terrain:FillBlock(CFrame.new(cursor + Vector3.new(14, -1.5, 0)), Vector3.new(16, 3, 8), Enum.Material.Cobblestone)
	local finish = B.cyl(obby, "Finish", summit + Vector3.new(0, 0.25, 0), 0.5, 10, B.palette.gold, Enum.Material.Neon, { CastShadow = false })
	finish.Transparency = 0.15
	B.sparkles(finish, B.palette.gold, 16, 0.5)
	local star = B.part(obby, {
		Name = "StarShard",
		Size = Vector3.new(4, 4, 4),
		CFrame = CFrame.new(summit + Vector3.new(0, 9, 0)) * CFrame.Angles(math.rad(45), math.rad(30), math.rad(45)),
		Color = B.palette.pink,
		Material = Enum.Material.Neon,
		CanCollide = false,
		CastShadow = false,
	})
	B.light(star, B.palette.pink, 30, 2)
	B.sparkles(star, B.palette.pink, 20, 0.8)
	B.billboard(star, "✦ ОСКОЛОК ЗВЕЗДЫ ✦", B.palette.pink, 5, 300, 200)
	for i = 1, 6 do
		local a = i / 6 * math.pi * 2
		B.crystalCluster(summit + Vector3.new(math.cos(a) * 15, 0, math.sin(a) * 15), i % 2 == 0 and B.palette.pink or B.palette.cyan)
	end
	B.note(summit + Vector3.new(-8, 0, 8), 5, summit)

	-- портал домой
	local portal = B.cyl(obby, "ReturnPortal", summit + Vector3.new(10, 4, -10), 0.6, 7, B.palette.violet, Enum.Material.Neon, { CanCollide = false, CastShadow = false })
	portal.CFrame = CFrame.new(summit + Vector3.new(10, 4, -10)) * CFrame.Angles(0, math.rad(45), 0)
	portal.Transparency = 0.25
	portal:SetAttribute("Teleport", "Hub")
	B.sparkles(portal, B.palette.violet, 20, 0.6)
	B.billboard(portal, "🏠 Домой", B.palette.violet, 5, 160, 60)
end
