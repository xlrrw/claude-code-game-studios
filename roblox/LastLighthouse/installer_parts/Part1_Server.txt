local profiles = {} -- [player] = { data = {}, canSave = bool, saving = bool }

local store = nil

	local ok, result = pcall(function()
		return DataStoreService:GetDataStore(Config.DataStoreName)
	end)
	if ok then
		store = result
	else
		warn("[Маяк] DataStore недоступен — прогресс не будет сохраняться. Включи: Game Settings → Security → Enable Studio Access to API Services. Ошибка: " .. tostring(result))
	end

local function deepCopy(value)
	if type(value) ~= "table" then
		return value
	end
	local copy = {}
	for k, v in pairs(value) do
		copy[k] = deepCopy(v)
	end
	return copy
end

local function reconcile(data, template)
	for key, value in pairs(template) do
		if data[key] == nil then
			data[key] = deepCopy(value)
		elseif type(value) == "table" and type(data[key]) == "table" then
			-- для словарей с фиксированными ключами (upgrades) добиваем недостающие
			if next(value) ~= nil and #value == 0 then
				reconcile(data[key], value)
			end
		end
	end
	return data
end

local function keyFor(player)
	return "player_" .. player.UserId
end

function DataService.Load(player)
	local data = nil
	local loadedOk = false

	if store then
		-- в Studio одна попытка, чтобы не ждать, если доступ к API выключен
		local attempts = RunService:IsStudio() and 1 or 4
		for attempt = 1, attempts do
			local ok, result = pcall(function()
				return store:GetAsync(keyFor(player))
			end)
			if ok then
				data = result
				loadedOk = true
				break
			end
			warn("[Маяк] Не удалось загрузить данные " .. player.Name .. " (попытка " .. attempt .. "): " .. tostring(result))
			if attempt < attempts then
				task.wait(attempt * 1.5)
			end
		end
	end

	if player.Parent == nil then
		return nil
	end

	if type(data) ~= "table" then
		data = deepCopy(TEMPLATE)
		data.firstJoin = os.time()
		if RunService:IsStudio() then
			data.light = Config.Debug.StartingLightInStudio or 0
		end
	end
	reconcile(data, TEMPLATE)

	profiles[player] = {
		data = data,
		-- если загрузка упала, НЕ сохраняем, иначе перезапишем настоящий прогресс пустым
		canSave = store ~= nil and loadedOk,
		saving = false,
	}
	if store and not loadedOk then
		warn("[Маяк] Прогресс " .. player.Name .. " не загрузился — в этой сессии он не будет сохраняться.")
	end

	loadedEvent:Fire(player, data)
	return data
end

function DataService.Get(player)
	local profile = profiles[player]
	return profile and profile.data or nil
end

-- Ждёт загрузки профиля (нужно для обработки покупок).
function DataService.WaitFor(player, timeout)
	local deadline = os.clock() + (timeout or 10)
	while os.clock() < deadline do
		local profile = profiles[player]
		if profile then
			return profile.data
		end
		if player.Parent == nil then
			return nil
		end
		task.wait(0.2)
	end
	return nil
end

function DataService.CanSave(player)
	local profile = profiles[player]
	return profile ~= nil and profile.canSave
end

-- Возвращает true, если сохранение прошло успешно.
function DataService.Save(player)
	local profile = profiles[player]
	if not profile or not profile.canSave then
		return false
	end
	while profile.saving do
		task.wait(0.1)
	end
	profile.saving = true
	local success = false
	for attempt = 1, 3 do
		local ok, err = pcall(function()
			store:UpdateAsync(keyFor(player), function()
				return profile.data
			end)
		end)
		if ok then
			success = true
			break
		end
		warn("[Маяк] Ошибка сохранения " .. player.Name .. " (попытка " .. attempt .. "): " .. tostring(err))
		task.wait(attempt)
	end
	profile.saving = false
	return success
end

function DataService.RecordPurchase(player, purchaseId)
	local data = DataService.Get(player)
	if not data then
		return
	end
	data.purchases[purchaseId] = true
	table.insert(data.purchaseLog, purchaseId)
	while #data.purchaseLog > MAX_PURCHASE_LOG do
		local oldest = table.remove(data.purchaseLog, 1)
		data.purchases[oldest] = nil
	end
end

function DataService.Release(player)
	DataService.Save(player)
	profiles[player] = nil
end

function DataService.Start()
	Players.PlayerRemoving:Connect(function(player)
		DataService.Release(player)
	end)

	task.spawn(function()
		while true do
			task.wait(AUTOSAVE_SECONDS)
			for player in pairs(profiles) do
				task.spawn(DataService.Save, player)
			end
		end
	end)

	game:BindToClose(function()
		if RunService:IsStudio() and not store then
			return
		end
		local pending = 0
		for player in pairs(profiles) do
			pending = pending + 1
			task.spawn(function()
				DataService.Save(player)
				pending = pending - 1
			end)
		end
		local deadline = os.clock() + 25

local function makeShardPart(spawnPart, value)
	local part = Instance.new("Part")
	part.Name = "LightShard"
	part.Size = value > Config.Light.ShardValue and Vector3.new(1.3, 1.3, 1.3) or Vector3.new(0.9, 0.9, 0.9)
	part.Material = Enum.Material.Neon
	if value > Config.Light.ShardValue then
		part.Color = Color3.fromRGB(255, 120, 220)
	else
		part.Color = SHARD_COLORS[rng:NextInteger(1, #SHARD_COLORS)]
	end
	part.Anchored = true
	part.CanCollide = false
	part.CanTouch = false
	part.CanQuery = false
	part.CastShadow = false
	part.CFrame = CFrame.new(spawnPart.Position) * CFrame.Angles(math.rad(45), 0, math.rad(45))
	part:SetAttribute("Value", value)

	local attachment = Instance.new("Attachment")
	attachment.Parent = part
	local sparkle = Instance.new("ParticleEmitter")
	sparkle.Texture = "rbxasset://textures/particles/sparkles_main.dds"
	sparkle.Color = ColorSequence.new(part.Color)
	sparkle.LightEmission = 1
	sparkle.Size = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.3),
		NumberSequenceKeypoint.new(1, 0),
	})
	sparkle.Transparency = NumberSequence.new(0.1, 1)
	sparkle.Lifetime = NumberRange.new(0.6, 1)
	sparkle.Rate = 5
	sparkle.Speed = NumberRange.new(0.5, 1.5)
	sparkle.SpreadAngle = Vector2.new(180, 180)
	sparkle.Parent = attachment

	return part
end

local function setupShards()
	activeFolder = world:FindFirstChild("ActiveShards")
	if not activeFolder then

local function collect(player, shard)
	shard.active = false
	local position = shard.part.Position
	shard.part.Parent = nil
	local amount = math.max(1, math.floor(shard.value * State.GetMultiplier(player) + 0.5))
	State.AddLight(player, amount)
	State.Remotes.Fx:FireClient(player, "Pickup", position, amount)
	task.delay(Config.Light.ShardRespawnSeconds, function()
		shard.active = true
		shard.part.Parent = activeFolder
	end)
end

local function pickupLoop()
	local accumulator = 0
	RunService.Heartbeat:Connect(function(dt)
		accumulator = accumulator + dt
		if accumulator < Config.Light.PickupCheckInterval then
			return
		end
		accumulator = 0
		for _, player in ipairs(Players:GetPlayers()) do
			local character = player.Character
			local root = character and character:FindFirstChild("HumanoidRootPart")
			local humanoid = character and character:FindFirstChildOfClass("Humanoid")
			if root and humanoid and humanoid.Health > 0 and DataService.Get(player) then
				local radius = State.GetRadius(player)
				local position = root.Position
				for _, shard in ipairs(shards) do
					if shard.active and (shard.part.Position - position).Magnitude <= radius then
						collect(player, shard)
					end
				end
			end
		end
	end)
end

local function recomputeLevel(announce)
	local thresholds = Config.Lighthouse.LevelThresholds
	local w = State.World
	local oldLevel = w.level
	local level = 1
	for i, threshold in ipairs(thresholds) do
		if w.lighthouseTotal >= threshold then
			level = i
		end
	end
	w.level = level
	w.prevThreshold = thresholds[level]
	w.nextThreshold = thresholds[level + 1]
	applyLighthouseVisuals()
	if level ~= oldLevel then
		State.PushAllPlayers()
		if announce and level > oldLevel then
			State.NotifyAll("🗼 Маяк разгорелся до уровня " .. level .. "! Весь сервер получает +" .. math.floor((level - 1) * Config.Lighthouse.BonusPerLevel * 100 + 0.5) .. "% света", "level")
			State.Remotes.Fx:FireAllClients("LighthouseLevel", level)
		end
	end
	State.PushWorld()
end

function LightService.Deposit(player, fraction)
	if type(fraction) ~= "number" then
		return
	end
	if fraction ~= 0.25 and fraction ~= 0.5 and fraction ~= 1 then
		return
	end
	local data = DataService.Get(player)
	if not data then
		return
	end
	local amount = math.floor(data.light * fraction)
	if amount < 1 then
		State.Notify(player, "У тебя пока нет света. Собирай сияющие осколки!", "warn")
		return
	end
	if not State.SpendLight(player, amount) then
		return
	end
	State.AddXp(player, amount * Config.Lighthouse.XpPerLight)
	State.World.lighthouseTotal = State.World.lighthouseTotal + amount

	local shadow = State.World.shadow
	if shadow.active then
		shadow.progress = shadow.progress + amount
		contributors[player] = true
	end

	State.Remotes.Fx:FireClient(player, "Deposit", amount)
	State.Notify(player, "Ты отдал Маяку " .. State.FormatNumber(amount) .. " ✦ и получил столько же опыта Смотрителя", "deposit")
	recomputeLevel(true)
end

local function runShadowEvent()
	local cfg = Config.ShadowEvent
	local shadow = State.World.shadow
	contributors = {}
	shadow.active = true
	shadow.progress = 0
	shadow.goal = math.max(cfg.GoalMin, #Players:GetPlayers() * cfg.GoalPerPlayer)
	shadow.endsAt = State.Now() + cfg.Duration
	State.NotifyAll("🌑 ТЕНЬ НАСТУПАЕТ! Вложите в Маяк " .. shadow.goal .. " ✦ за " .. math.floor(cfg.Duration / 60) .. " минуты!", "shadow")
	State.Remotes.Fx:FireAllClients("ShadowStart")
	State.PushWorld()

	while State.Now() < shadow.endsAt and shadow.progress < shadow.goal do
		task.wait(0.5)
		State.PushWorld()
	end

	shadow.active = false
	if shadow.progress >= shadow.goal then
		State.World.shadowBoostUntil = State.Now() + cfg.SuccessBoostMinutes * 60
		State.NotifyAll("☀️ Тень отступила! Весь сервер получает +" .. math.floor(cfg.SuccessBonus * 100) .. "% света на " .. cfg.SuccessBoostMinutes .. " минут!", "victory")
		for player in pairs(contributors) do
			if player.Parent then
				State.AddLight(player, cfg.RewardLight)
				State.AddXp(player, cfg.RewardLight)
				State.Notify(player, "Награда героя: +" .. cfg.RewardLight .. " ✦", "purchase")
			end
		end
		State.Remotes.Fx:FireAllClients("ShadowEnd", true)
		task.delay(cfg.SuccessBoostMinutes * 60 + 1, function()
			State.PushAllPlayers()
			State.PushWorld()
		end)
	else
		State.World.lighthouseTotal = math.floor(State.World.lighthouseTotal * (1 - cfg.FailPenalty))
		State.NotifyAll("💨 Тень погасила часть огня... Маяк потускнел. В следующий раз — вместе!", "shadow")
		State.Remotes.Fx:FireAllClients("ShadowEnd", false)
	end
	contributors = {}
	State.PushAllPlayers()
	recomputeLevel(false)
end

local function shadowLoop()
	local cfg = Config.ShadowEvent

		local delaySeconds = cfg.FirstDelaySeconds

			State.PushWorld()
			task.wait(delaySeconds)
			if #Players:GetPlayers() > 0 then
				runShadowEvent()
			end
			delaySeconds = rng:NextInteger(cfg.IntervalMin, cfg.IntervalMax)
		end
	end)
end

local function processReceipt(receiptInfo)
	local player = Players:GetPlayerByUserId(receiptInfo.PlayerId)
	if not player then
		-- игрок вышел — Roblox повторит, когда он зайдёт снова
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local data = DataService.WaitFor(player, 15)
	if not data then
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	if data.purchases[receiptInfo.PurchaseId] then
		return Enum.ProductPurchaseDecision.PurchaseGranted
	end

	local key = productKeyById[receiptInfo.ProductId]
	local handler = key and handlers[key]
	if not handler then
		warn("[Маяк] Неизвестный товар " .. tostring(receiptInfo.ProductId) .. " — проверь MonetizationIds")
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	local ok, err = pcall(handler, player, data)
	if not ok then
		warn("[Маяк] Ошибка выдачи товара " .. key .. ": " .. tostring(err))
		return Enum.ProductPurchaseDecision.NotProcessedYet
	end

	DataService.RecordPurchase(player, receiptInfo.PurchaseId)

	if DataService.CanSave(player) then
		if not DataService.Save(player) then
			-- запись о покупке осталась в памяти: при повторе Roblox товар не выдастся дважды
			return Enum.ProductPurchaseDecision.NotProcessedYet
		end
	end
	return Enum.ProductPurchaseDecision.PurchaseGranted
end

local function setupCheckpoints()
	local folder = obby and obby:FindFirstChild("Checkpoints")
	if not folder then
		return
	end
	for _, cp in ipairs(folder:GetChildren()) do
		if cp:IsA("BasePart") then
			cp.Touched:Connect(function(hit)
				local player = playerFromHit(hit)
				if not player then
					return
				end
				local current = checkpoints[player]
				local index = cp:GetAttribute("Index") or 0
				if current == cp then
					return
				end
				if current and (current:GetAttribute("Index") or 0) >= index then
					return
				end
				checkpoints[player] = cp
				State.Remotes.Fx:FireClient(player, "Checkpoint", cp.Position, index)
			end)
		end
	end
end

local function setupFalling()
	local folder = obby and obby:FindFirstChild("FallingPlatforms")
	if not folder then
		return
	end
	for _, platform in ipairs(folder:GetChildren()) do
		if platform:IsA("BasePart") then
			local origin = platform.CFrame
			local originColor = platform.Color
			local originTransparency = platform.Transparency
			-- светящийся обод (дочерняя деталь Rim) двигаем вместе с платформой
			local rim = platform:FindFirstChild("Rim")
			local rimOffset = rim and origin:ToObjectSpace(rim.CFrame)
			local rimTransparency = rim and rim.Transparency
			local function moveRim(cf, transparency, time)
				if not rim then
					return
				end
				if time then
					TweenService:Create(rim, TweenInfo.new(time, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
						CFrame = cf * rimOffset,
						Transparency = transparency,
					}):Play()
				else
					rim.CFrame = cf * rimOffset
				end
			end
			local busy = false
			platform.Touched:Connect(function(hit)
				if busy or not playerFromHit(hit) then
					return
				end
				busy = true
				platform.Color = Color3.fromRGB(255, 110, 140)
				-- дрожь
				local shakes = math.max(2, math.floor(Config.Obby.FallWarnSeconds / 0.08))
				for i = 1, shakes do
					local shake = origin * CFrame.new((i % 2 == 0 and 0.12 or -0.12), 0, 0)
					platform.CFrame = shake
					moveRim(shake)
					task.wait(0.08)
				end
				platform.CanCollide = false
				local fallen = origin * CFrame.new(0, -25, 0) * CFrame.Angles(math.rad(25), 0, math.rad(15))
				TweenService:Create(platform, TweenInfo.new(0.9, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
					CFrame = fallen,
					Transparency = 1,
				}):Play()
				moveRim(fallen, 1, 0.9)
				task.wait(Config.Obby.FallRespawnSeconds)
				platform.CFrame = origin * CFrame.new(0, -3, 0)
				moveRim(origin * CFrame.new(0, -3, 0))
				platform.Color = originColor
				platform.CanCollide = true
				TweenService:Create(platform, TweenInfo.new(0.4, Enum.EasingStyle.Back), {
					CFrame = origin,
					Transparency = originTransparency,
				}):Play()
				if rim then
					TweenService:Create(rim, TweenInfo.new(0.4, Enum.EasingStyle.Back), {
						CFrame = origin * rimOffset,
						Transparency = rimTransparency,
					}):Play()
				end
				task.wait(0.4)
				busy = false
			end)
		end
	end
end

local function setupBlink()
	local folder = obby and obby:FindFirstChild("BlinkPlatforms")
	if not folder then
		return
	end
	local groups = { [0] = {}, [1] = {} }
	for _, platform in ipairs(folder:GetChildren()) do
		if platform:IsA("BasePart") then
			local phase = platform:GetAttribute("Phase") or 0
			table.insert(groups[phase % 2], platform)
		end
	end
	task.spawn(function()
		local phase = 0
		while true do
			for groupPhase, list in pairs(groups) do
				local solid = groupPhase == phase
				for _, platform in ipairs(list) do
					platform.CanCollide = solid
					platform.Transparency = solid and 0 or 0.8
					platform.Material = solid and Enum.Material.Neon or Enum.Material.ForceField
				end
			end
			-- предупреждение: за 0.5 сек до смены твёрдые платформы мерцают
			task.wait(Config.Obby.BlinkPeriodSeconds - 0.5)
			for _, platform in ipairs(groups[phase]) do
				platform.Transparency = 0.4
			end
			task.wait(0.5)
			phase = 1 - phase
		end
	end)
end

local function setupFinish()
	local finish = obby and obby:FindFirstChild("Finish", true)
	if not finish or not finish:IsA("BasePart") then
		return
	end
	finish.Touched:Connect(function(hit)
		local player = playerFromHit(hit)
		if not player then
			return
		end
		local now = os.clock()
		if lastFinish[player] and now - lastFinish[player] < Config.Obby.FinishCooldownSeconds then
			return
		end
		lastFinish[player] = now
		local data = DataService.Get(player)
		if not data then
			return
		end
		data.obbyFinishes = data.obbyFinishes + 1
		local reward = math.floor(Config.Obby.FinishReward * State.GetMultiplier(player) + 0.5)
		State.AddLight(player, reward)
		State.AddXp(player, Config.Obby.FinishXp)
		State.Remotes.Fx:FireClient(player, "ObbyFinish", reward)
		if data.starShards < 1 then
			data.starShards = 1
			State.Remotes.Dialogue:FireClient(player, {
				speaker = "Осколок Звезды",
				lines = { Config.Story.StarShardMessage },
			})
		else
			State.Notify(player, "Паркур пройден! +" .. State.FormatNumber(reward) .. " ✦", "victory")
		end
		State.PushPlayer(player)
		task.delay(3, function()
			if player.Parent then
				ObbyService.ReturnToHub(player)
			end
		end)
	end)
end

local function setupPortals()
	-- любые детали с атрибутом Teleport = "Hub" возвращают на остров с Маяком
	for _, inst in ipairs(world:GetDescendants()) do
		if inst:IsA("BasePart") and inst:GetAttribute("Teleport") == "Hub" then
			local debounce = {}
			inst.Touched:Connect(function(hit)
				local player = playerFromHit(hit)
				if player and not debounce[player] then
					debounce[player] = true
					ObbyService.ReturnToHub(player)
					task.delay(1, function()
						debounce[player] = nil
					end)
				end
			end)
		end
	end
end

local function voidLoop()
	task.spawn(function()
		while true do
			task.wait(0.25)
			for _, player in ipairs(Players:GetPlayers()) do
				local character = player.Character
				local root = character and character:FindFirstChild("HumanoidRootPart")
				if root and root.Position.Y < Config.Obby.VoidY then
					local cp = checkpoints[player]
					if cp and cp.Parent then
						teleport(player, cp.CFrame + Vector3.new(0, 4, 0))
					else
						teleport(player, hubSpawnCFrame())
					end
					State.Remotes.Fx:FireClient(player, "Respawned")
				end
			end
		end
	end)
end

	setupFalling()
	setupBlink()
	setupFinish()
	setupPortals()
	voidLoop()
	Players.PlayerRemoving:Connect(function(player)
		checkpoints[player] = nil
		lastFinish[player] = nil
	end)
end

	local nextRank = Config.Ranks[index + 1]
	return index, Config.Ranks[index], nextRank
end

function State.GetMultiplier(player)
	local data = DataService.Get(player)
	local passes = State.Passes[player] or {}
	local now = State.Now()
	local mult = 1

	if passes.VIP then
		mult = mult + Config.Passes.VIP.lightBonus
	end
	if player.MembershipType == Enum.MembershipType.Premium then
		mult = mult + Config.Premium.LightBonus
	end
	mult = mult + (State.World.level - 1) * Config.Lighthouse.BonusPerLevel
	if State.World.globalBoostUntil > now then
		mult = mult + Config.Products.GlobalBoost.bonus
	end
	if State.World.shadowBoostUntil > now then
		mult = mult + Config.ShadowEvent.SuccessBonus
	end
	if data then
		local rankIndex = State.GetRank(data.keeperXp)
		mult = mult + (rankIndex - 1) * Config.RankBonus
		for _, petId in ipairs(data.equipped) do
			local pet = Config.PetById[petId]
			if pet then
				mult = mult + pet.lightBonus
			end
		end
	end
	return mult
end

function State.GetRadius(player)
	local data = DataService.Get(player)
	local radius = Config.Light.BasePickupRadius
	if data then
		radius = radius + (data.upgrades.Radius or 0) * Config.Upgrades.Radius.perLevel
		for _, petId in ipairs(data.equipped) do
			local pet = Config.PetById[petId]
			if pet then
				radius = radius + pet.radiusBonus
			end
		end
	end
	return radius
end

function State.GetWalkSpeed(player)
	local data = DataService.Get(player)
	local level = data and data.upgrades.Speed or 0
	return 16 + level * Config.Upgrades.Speed.perLevel
end

function State.GetSlots(player)
	local passes = State.Passes[player] or {}
	if passes.PetSlots then
		return Config.PassEquipSlots
	end
	return Config.BaseEquipSlots
end

function State.PushPlayer(player)
	if pendingPlayers[player] then
		return

		pendingPlayers[player] = nil

		end

		if not data then
			return
		end
		State.Remotes.PlayerState:FireClient(player, buildSnapshot(player, data))
		State.UpdateLeaderstats(player, data)
	end)
end

local function formatNumber(n)
	n = math.floor(n)
	if n >= 1000000 then
		return string.format("%.1fM", n / 1000000)
	elseif n >= 10000 then
		return string.format("%.1fK", n / 1000)
	end
	return tostring(n)
end

	local node = game
	for part in string.gmatch(path, "[^%.]+") do
		if node == game then
			node = game:GetService(part)
		else
			node = node:FindFirstChild(part) or node:WaitForChild(part, 5)
		end
	end
	return node
end
