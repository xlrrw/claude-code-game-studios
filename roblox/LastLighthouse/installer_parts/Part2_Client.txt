-- Осколки: парят и вращаются
------------------------------------------------------------------------
local function trackShard(part)
	if part:IsA("BasePart") then
		shardBase[part] = { position = part.Position, phase = math.random() * math.pi * 2 }
	end
end

local function animateShards(t)
	local folder = world:FindFirstChild("ActiveShards")
	if not folder then
		return
	end
	for _, part in ipairs(folder:GetChildren()) do
		local base = shardBase[part]
		if not base and part:IsA("BasePart") then
			trackShard(part)
			base = shardBase[part]
		end
		if base then
			local y = math.sin(t * 2 + base.phase) * 0.45
			part.CFrame = CFrame.new(base.position + Vector3.new(0, y, 0)) * CFrame.Angles(t * 0.9 + base.phase, t * 1.4, math.rad(45))
		end
	end
end

------------------------------------------------------------------------
-- Маяк: вращающийся луч

local beamPivot, beamPivotBase
local beams = {}

local function setupLighthouse()
	local lighthouse = world:FindFirstChild("Lighthouse")
	if not lighthouse then
		return
	end
	beamPivot = lighthouse:FindFirstChild("BeamPivot", true)
	if beamPivot then
		beamPivotBase = beamPivot.CFrame
		for _, child in ipairs(beamPivot:GetDescendants()) do
			if child:IsA("Beam") then
				table.insert(beams, { beam = child, width = child.Width1 })
			end
		end
	end
	local function applyLevel()
		local level = lighthouse:GetAttribute("Level") or 1
		local k = 0.6 + level * 0.12
		for _, b in ipairs(beams) do
			b.beam.Width1 = b.width * k
		end
	end
	lighthouse:GetAttributeChangedSignal("Level"):Connect(applyLevel)
	applyLevel()
end

local function animateLighthouse(t)
	if beamPivot and beamPivotBase then
		local speed = shadowActive and 0.25 or 0.55
		beamPivot.CFrame = beamPivotBase * CFrame.Angles(0, t * speed, 0)
	end
end

-- Северное сияние

local aurora = {}
local function setupAurora()
	local sky = world:FindFirstChild("Sky")
	if not sky then
		return
	end
	for _, b in ipairs(sky:GetDescendants()) do
		if b:IsA("Beam") then
			table.insert(aurora, { beam = b, c0 = b.CurveSize0, c1 = b.CurveSize1, phase = math.random() * 10 })
		end
	end
end

local function animateAurora(t)
	for _, a in ipairs(aurora) do
		a.beam.CurveSize0 = a.c0 + math.sin(t * 0.25 + a.phase) * 40
		a.beam.CurveSize1 = a.c1 + math.cos(t * 0.2 + a.phase) * 40
	end
end

-- Тень, кружащая вдали

local wisp, wispRadius = nil, 300
local function animateWisp(t, dt)
	if not wisp then
		wisp = world:FindFirstChild("ShadowWisp")
		if not wisp then
			return
		end
	end
	local targetRadius = shadowActive and 110 or 300
	wispRadius = wispRadius + (targetRadius - wispRadius) * math.min(1, dt * 0.5)
	local angle = t * 0.05
	local height = 40 + math.sin(t * 0.3) * 12
	local position = Vector3.new(math.cos(angle) * wispRadius, height, math.sin(angle) * wispRadius)
	wisp:PivotTo(CFrame.lookAt(position, Vector3.new(0, 30, 0)))
end

------------------------------------------------------------------------
-- Духи: лёгкое покачивание

local function animatePets(t)
	local root = getRoot()
	if not root then
		return
	end
	for i = 1, 3 do
		local slot = root:FindFirstChild("PetSlot" .. i)
		if slot then
			local base = slot:GetAttribute("BasePosition")
			if base then
				slot.Position = base + Vector3.new(0, math.sin(t * 3 + i * 1.7) * 0.4, 0)
			end
		end
	end
end

local function burst(position, color, count, size)
	local part = Instance.new("Part")
	part.Anchored = true
	part.CanCollide = false
	part.CanQuery = false
	part.CanTouch = false
	part.Transparency = 1
	part.Size = Vector3.new(0.2, 0.2, 0.2)
	part.Position = position
	part.Parent = workspace
	local emitter = Instance.new("ParticleEmitter")
	emitter.Texture = "rbxasset://textures/particles/sparkles_main.dds"
	emitter.Color = ColorSequence.new(color)
	emitter.LightEmission = 1
	emitter.Size = NumberSequence.new({
		NumberSequenceKeypoint.new(0, size or 0.5),
		NumberSequenceKeypoint.new(1, 0),
	})
	emitter.Transparency = NumberSequence.new(0, 1)
	emitter.Lifetime = NumberRange.new(0.5, 1.1)
	emitter.Speed = NumberRange.new(6, 14)
	emitter.SpreadAngle = Vector2.new(180, 180)
	emitter.Drag = 4
	emitter.Rate = 0
	emitter.Parent = part
	emitter:Emit(count or 20)
	task.delay(1.5, function()
		part:Destroy()
	end)
end

function Effects.Pickup(position, amount)
	Sound.Play("Pickup")
	burst(position, Color3.fromRGB(255, 230, 140), 12, 0.35)
	-- маленький огонёк летит к игроку
	local root = getRoot()
	if root then
		local orb = Instance.new("Part")
		orb.Shape = Enum.PartType.Ball
		orb.Size = Vector3.new(0.5, 0.5, 0.5)
		orb.Material = Enum.Material.Neon
		orb.Color = Color3.fromRGB(255, 230, 140)
		orb.Anchored = true
		orb.CanCollide = false
		orb.CanQuery = false
		orb.CanTouch = false
		orb.Position = position
		orb.Parent = workspace
		local start = os.clock()
		local conn
		conn = RunService.RenderStepped:Connect(function()
			local alpha = math.min(1, (os.clock() - start) / 0.35)
			local r = getRoot()
			if not r or alpha >= 1 then
				conn:Disconnect()
				orb:Destroy()
				return
			end
			orb.Position = position:Lerp(r.Position, alpha * alpha)
		end)
	end
	floatingText(position + Vector3.new(0, 1.5, 0), "+" .. amount .. " ✦", Color3.fromRGB(255, 220, 110))
end

		task.delay(i * 0.08, function()

end

function Effects.ScreenFlash(color, strength)
	local cc = lighting.color
	if not cc then
		return
	end
	local base = lighting.colorBase
	cc.Brightness = base.Brightness + (strength or 0.3)
	cc.TintColor = color or Color3.new(1, 1, 1)
	tween(cc, 0.8, { Brightness = shadowActive and -0.05 or base.Brightness, TintColor = shadowActive and Color3.fromRGB(190, 160, 230) or base.TintColor })
end

------------------------------------------------------------------------
-- Атмосфера при нашествии Тени
------------------------------------------------------------------------
local function applyShadowMood(active)
	if active == shadowActive then
		return
	end
	shadowActive = active
	local cc, atmo = lighting.color, lighting.atmosphere
	if cc then
		if active then
			tween(cc, 3, { TintColor = Color3.fromRGB(190, 160, 230), Saturation = -0.15, Brightness = -0.05 })
		else
			tween(cc, 3, { TintColor = lighting.colorBase.TintColor, Saturation = lighting.colorBase.Saturation, Brightness = lighting.colorBase.Brightness })
		end
	end
	if atmo then
		if active then
			tween(atmo, 3, { Density = math.min(0.75, lighting.atmoBase.Density + 0.25), Color = Color3.fromRGB(60, 30, 90) })
		else
			tween(atmo, 3, { Density = lighting.atmoBase.Density, Color = lighting.atmoBase.Color })
		end
	end
	if active then
		Sound.Play("Shadow")
	end
end

function Effects.SetWorldState(ws)
	applyShadowMood(ws.shadow and ws.shadow.active == true)
end

------------------------------------------------------------------------
function Effects.Init()
	local cc = Lighting:FindFirstChild("LL_Color")
	local atmo = Lighting:FindFirstChildOfClass("Atmosphere")
	if cc then
		lighting.color = cc
		lighting.colorBase = { TintColor = cc.TintColor, Saturation = cc.Saturation, Brightness = cc.Brightness }
	end
	if atmo then
		lighting.atmosphere = atmo
		lighting.atmoBase = { Density = atmo.Density, Color = atmo.Color }
	end

	setupLighthouse()
	setupAurora()

	RunService.RenderStepped:Connect(function(dt)
		local t = os.clock()
		animateShards(t)
		animateLighthouse(t)
		animateAurora(t)
		animateWisp(t, dt)
		animatePets(t)
	end)

	Sound.StartMusic()
end

return Effects

local function setupJumpPads()
	local world = workspace:WaitForChild("LastLighthouseWorld")
	local obby = world:FindFirstChild("Obby")
	local pads = obby and obby:FindFirstChild("JumpPads")
	if not pads then
		return
	end
	for _, pad in ipairs(pads:GetChildren()) do
		if pad:IsA("BasePart") then
			pad.Touched:Connect(function(hit)
				local character = player.Character
				if not character or not hit:IsDescendantOf(character) then
					return
				end
				if os.clock() - lastPad < 0.6 then
					return
				end
				lastPad = os.clock()
				local root = character:FindFirstChild("HumanoidRootPart")
				local humanoid = character:FindFirstChildOfClass("Humanoid")
				if root and humanoid then
					local v = root.AssemblyLinearVelocity
					root.AssemblyLinearVelocity = Vector3.new(v.X, Config.Obby.JumpPadVelocity, v.Z)
					humanoid:ChangeState(Enum.HumanoidStateType.Freefall)
					usedDoubleJump = false
					if effects then
						effects.Burst(pad.Position + Vector3.new(0, 1, 0), Color3.fromRGB(110, 240, 160), 30, 0.7)
						effects.Ring(pad.Position + Vector3.new(0, 0.5, 0), Color3.fromRGB(110, 240, 160))
					end
				end
			end)
		end
	end
end

	effects = effectsModule
	if player.Character then
		task.spawn(onCharacter, player.Character)
	end
	player.CharacterAdded:Connect(onCharacter)

	UserInputService.JumpRequest:Connect(function()
		if not player:GetAttribute("HasWings") or usedDoubleJump then
			return
		end
		local character = player.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not humanoid or not root then
			return
		end
		if humanoid:GetState() ~= Enum.HumanoidStateType.Freefall then
			return
		end
		usedDoubleJump = true
		local v = root.AssemblyLinearVelocity
		root.AssemblyLinearVelocity = Vector3.new(v.X, 55, v.Z)
		humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
		wingBurst(root)
	end)

	task.spawn(setupJumpPads)
end

return Movement

local function new(className, props, children)
	local inst = Instance.new(className)
	local parent = nil
	if props then
		for k, v in pairs(props) do
			if k == "Parent" then
				parent = v
			else
				inst[k] = v
			end
		end
	end
	if children then
		for _, child in ipairs(children) do
			child.Parent = inst
		end
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function formatNumber(n)
	n = math.floor(n or 0)
	if n >= 1000000 then
		return string.format("%.2fM", n / 1000000)
	elseif n >= 10000 then
		return string.format("%.1fK", n / 1000)
	end
	local s = tostring(n)
	local formatted = s:reverse():gsub("(%d%d%d)", "%1 "):reverse()
	return (formatted:gsub("^ ", ""))
end
UI.FormatNumber = formatNumber

local function formatTime(seconds)
	seconds = math.max(0, math.floor(seconds))
	return string.format("%d:%02d", math.floor(seconds / 60), seconds % 60)
end

function UI.OpenWindow(obj)
	if openWindow and openWindow ~= obj then
		UI.CloseWindow(openWindow, true)
	end
	openWindow = obj
	obj.overlay.Visible = true
	obj.overlay.BackgroundTransparency = 1
	obj.scale.Scale = 0.75
	tween(obj.overlay, 0.2, { BackgroundTransparency = 0.45 })
	tween(obj.scale, 0.35, { Scale = 1 }, Enum.EasingStyle.Back)
end

function UI.CloseWindow(obj, instant)
	if openWindow == obj then
		openWindow = nil
	end
	if instant then
		obj.overlay.Visible = false
		return
	end
	tween(obj.overlay, 0.15, { BackgroundTransparency = 1 })
	local t = tween(obj.scale, 0.15, { Scale = 0.8 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
	t.Completed:Connect(function()
		if openWindow ~= obj then
			obj.overlay.Visible = false
		end
	end)
end

local function renderUpgrades(list)
	for order, id in ipairs(Config.UpgradeOrder) do
		local def = Config.Upgrades[id]
		local level = ps and ps.upgrades[id] or 0
		local maxed = level >= def.maxLevel
		local cost = Config.GetUpgradeCost(id, level)
		local c = card(list, order, C.Cyan)
		cardIcon(c, def.icon, C.Cyan)
		local effect
		if id == "Radius" then
			effect = "Сейчас: радиус " .. string.format("%.1f", ps and ps.radius or Config.Light.BasePickupRadius)
		else
			effect = "Сейчас: скорость " .. string.format("%.1f", 16 + level * def.perLevel)
		end
		cardText(c, def.name .. "  ·  " .. level .. "/" .. def.maxLevel, C.Text, def.description, effect)
		if maxed then
			cardButton(c, "МАКС ✓", C.Green, C.Cyan, C.Night, function() end)
		else
			local canAfford = ps and ps.light >= cost
			local b = cardButton(c, formatNumber(cost) .. " ✦", C.Gold, C.Gold2, C.Night, function()
				send("BuyUpgrade", id)
			end)
			if not canAfford then
				setButtonColors(b, Color3.fromRGB(110, 100, 150), Color3.fromRGB(70, 60, 110))
				b.TextColor3 = C.Text
			end
		end
	end
end

local function isEquipped(petId)

	for _, id in ipairs(ps.equipped) do

	end
	return false
end

local function playDialogue(entry)
	local style = DIALOGUE_STYLES[entry.style or "default"] or DIALOGUE_STYLES.default
	dialogue.gradient.Color = ColorSequence.new(style.c1, style.c2)
	dialogue.stroke.Color = style.accent
	dialogue.nameTagGradient.Color = ColorSequence.new(style.name, style.accent)
	dialogue.name.Text = entry.speaker or ""
	local box = dialogue.box
	box.Visible = true
	dialogue.scale.Scale = 0.85
	tween(dialogue.scale, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)

	for index, line in ipairs(entry.lines or {}) do
		dialogue.text.Text = line
		dialogue.text.MaxVisibleGraphemes = 0
		dialogue.hint.Text = index < #entry.lines and "Нажми, чтобы продолжить ▶" or "Нажми, чтобы закрыть ✓"
		local skipped = false
		local conn = dialogue.advance.Event:Connect(function()
			skipped = true
		end)
		local length = utf8.len(line) or #line
		local shown = 0
		while shown < length and not skipped do
			shown = shown + 1
			dialogue.text.MaxVisibleGraphemes = shown
			task.wait(0.022)
		end
		conn:Disconnect()
		dialogue.text.MaxVisibleGraphemes = -1
		dialogue.advance.Event:Wait()
	end
	local t = tween(dialogue.scale, 0.15, { Scale = 0.85 })
	t.Completed:Wait()
	box.Visible = false
end

-- Показывает диалог (в очереди). Если передан done (BindableEvent) — сработает по окончании.
function UI.ShowDialogue(entry, done)
	table.insert(dialogue.queue, { entry = entry, done = done })
	if dialogue.running then
		return
	end
	dialogue.running = true
	task.spawn(function()
		while #dialogue.queue > 0 do
			local item = table.remove(dialogue.queue, 1)
			playDialogue(item.entry)
			if item.done then
				item.done:Fire()
			end
		end
		dialogue.running = false
	end)
end

function UI.ShowDialogueAsync(entry)
	local done = Instance.new("BindableEvent")
	UI.ShowDialogue(entry, done)
	done.Event:Wait()
	done:Destroy()
end

local function buildToasts()
	toastHolder = new("Frame", {
		Name = "Toasts",
		Size = UDim2.fromOffset(340, 400),
		Position = UDim2.new(1, -16, 0, 16),
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		ZIndex = 50,
		Parent = gui,
	}, {
		new("UIListLayout", {
			Padding = UDim.new(0, 8),
			HorizontalAlignment = Enum.HorizontalAlignment.Right,
			SortOrder = Enum.SortOrder.LayoutOrder,
		}),
	})
	bannerHolder = new("Frame", {
		Name = "Banner",
		Size = UDim2.fromOffset(700, 140),
		Position = UDim2.fromScale(0.5, 0.32),
		AnchorPoint = Vector2.new(0.5, 0.5),
		BackgroundTransparency = 1,
		ZIndex = 60,
		Parent = gui,
	})
end

local toastCount = 0
function UI.Notify(text, kind)
	if not toastHolder then
		return
	end
	toastCount = toastCount + 1
	local color = KIND_COLORS[kind] or C.Cyan
	local toast = new("Frame", {
		Size = UDim2.fromOffset(340, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		BackgroundColor3 = Color3.new(1, 1, 1),
		LayoutOrder = toastCount,
		ZIndex = 50,
		Parent = toastHolder,
	}, { corner(14), gradient(C.Panel2, C.Night, 90), stroke(color, 2, 0.1), padding(10) })
	local stripe = new("Frame", {
		Size = UDim2.new(0, 5, 1, 0),
		BackgroundColor3 = color,
		ZIndex = 51,
		Parent = toast,
	}, { corner(3) })
	stripe.Position = UDim2.fromOffset(-4, 0)
	local l = label({
		Text = text,
		Size = UDim2.new(1, -8, 0, 0),
		Position = UDim2.fromOffset(8, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		TextSize = 16,
		TextWrapped = true,
		Font = Enum.Font.GothamBold,
		ZIndex = 51,
		Parent = toast,
	})
	l.TextColor3 = C.Text
	local s = new("UIScale", { Scale = 0.6, Parent = toast })
	tween(s, 0.3, { Scale = 1 }, Enum.EasingStyle.Back)
	task.delay(4.5, function()
		local t = tween(s, 0.25, { Scale = 0 }, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
		t.Completed:Wait()
		toast:Destroy()
	end)
	-- не больше 5 уведомлений сразу
	local toasts = {}
	for _, child in ipairs(toastHolder:GetChildren()) do
		if child:IsA("Frame") then
			table.insert(toasts, child)
		end
	end
	if #toasts > 5 then
		table.sort(toasts, function(a, b)
			return a.LayoutOrder < b.LayoutOrder
		end)
		toasts[1]:Destroy()
	end
end

function UI.Banner(title, subtitle, color)
	if not bannerHolder then
		return
	end
	bannerHolder:ClearAllChildren()
	local s = new("UIScale", { Scale = 0.3, Parent = bannerHolder })
	local t = label({
		Text = title,
		Size = UDim2.new(1, 0, 0, 80),
		TextSize = 64,
		TextXAlignment = Enum.TextXAlignment.Center,
		TextColor3 = Color3.new(1, 1, 1),
		ZIndex = 60,
		Parent = bannerHolder,
	})
	new("UIGradient", { Color = ColorSequence.new(color or C.Gold, C.Pink), Rotation = 90, Parent = t })
	textStroke(C.Night, 3).Parent = t
	local st = label({
		Text = subtitle or "",
		Size = UDim2.new(1, 0, 0, 36),
		Position = UDim2.fromOffset(0, 80),
		TextSize = 26,
		TextXAlignment = Enum.TextXAlignment.Center,
		TextColor3 = C.Text,
		ZIndex = 60,
		Parent = bannerHolder,
	})
	textStroke(C.Night, 2).Parent = st
	tween(s, 0.45, { Scale = 1 }, Enum.EasingStyle.Back)
	task.delay(2.6, function()
		if s.Parent then
			local out = tween(s, 0.3, { Scale = 0 }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
			out.Completed:Wait()
			if s.Parent then
				bannerHolder:ClearAllChildren()
			end
		end
	end)
end

function UI.BumpLight()
	if not hud.lightScale then
		return
	end
	hud.lightScale.Scale = 1.08
	tween(hud.lightScale, 0.25, { Scale = 1 }, Enum.EasingStyle.Back)
end

function UI.SetHudVisible(visible)
	for _, name in ipairs({ "LightPill", "RankPanel", "Lighthouse", "Boosts", "SideButtons", "Toasts" }) do
		local f = gui:FindFirstChild(name)
		if f then
			f.Visible = visible
		end
	end
	if hud.shadow then
		hud.shadow.Visible = visible and ws ~= nil and ws.shadow.active
	end
end

local function renderHud()
	if not ps then
		return
	end
	-- плавный счётчик света
	if lightTweenValue then
		tween(lightTweenValue, 0.5, { Value = ps.light })
	end
	hud.multLabel.Text = "звёздный свет · x" .. string.format("%.2f", ps.multiplier)
	hud.rankLabel.Text = "⭐ " .. ps.rankName
	local rank = Config.Ranks[ps.rankIndex]
	if rank then
		hud.rankLabel.TextColor3 = rank.color
	end
	if ps.nextRankXp then
		local ratio = (ps.keeperXp - ps.rankXp) / math.max(1, ps.nextRankXp - ps.rankXp)
		setProgress(hud.xpFill, ratio)
		hud.xpText.Text = formatNumber(ps.keeperXp) .. " / " .. formatNumber(ps.nextRankXp) .. " опыта"
	else
		setProgress(hud.xpFill, 1)
		hud.xpText.Text = "Максимальный ранг!"
	end
end

local function renderWorld()
	if not ws then
		return
	end
	hud.lighthouseLabel.Text = "🗼 МАЯК · уровень " .. ws.level .. (ws.level > 1 and ("  (+" .. math.floor((ws.level - 1) * Config.Lighthouse.BonusPerLevel * 100 + 0.5) .. "% всем)") or "")
	if ws.nextThreshold then
		local ratio = (ws.lighthouseTotal - ws.prevThreshold) / math.max(1, ws.nextThreshold - ws.prevThreshold)
		setProgress(hud.lhFill, ratio)
		hud.lhText.Text = formatNumber(ws.lighthouseTotal) .. " / " .. formatNumber(ws.nextThreshold) .. " ✦ до следующего уровня"
	else
		setProgress(hud.lhFill, 1)
		hud.lhText.Text = "Маяк сияет в полную силу!"
	end
	local shadow = ws.shadow
	local hudVisible = gui:FindFirstChild("LightPill") and gui.LightPill.Visible
	hud.shadow.Visible = shadow.active and hudVisible
	if shadow.active then
		setProgress(hud.shadowFill, shadow.progress / math.max(1, shadow.goal))
	end
end

-- таймеры (обновляются 4 раза в секунду)
local function timerLoop()
	local acc = 0
	RunService.Heartbeat:Connect(function(dt)
		acc = acc + dt
		if acc < 0.25 or not ws then
			return
		end
		acc = 0
		local now = workspace:GetServerTimeNow()
		local g = ws.globalBoostUntil - now
		hud.globalChip.Visible = g > 0
		if g > 0 then
			hud.globalChipText.Text = "🔥 x2 свет от " .. ws.globalBoostBy .. " · " .. formatTime(g)
		end
		local v = ws.shadowBoostUntil - now
		hud.victoryChip.Visible = v > 0
		if v > 0 then
			hud.victoryChipText.Text = "☀️ Победа над Тенью · " .. formatTime(v)
		end
		if ws.shadow.active then
			hud.shadowText.Text = formatNumber(ws.shadow.progress) .. " / " .. formatNumber(ws.shadow.goal) .. " ✦   ·   осталось " .. formatTime(ws.shadow.endsAt - now)
			hud.shadowStroke.Transparency = 0.5 + math.sin(os.clock() * 6) * 0.5
		end
	end)
end

function UI.Init(options)
	send = options.send
	local playerGui = player:WaitForChild("PlayerGui")

	scaleObj = new("UIScale", { Parent = gui })
	local function rescale()
		local camera = workspace.CurrentCamera
		if not camera then
			return
		end
		local vp = camera.ViewportSize
		scaleObj.Scale = math.clamp(math.min(vp.X / 1150, vp.Y / 680), 0.5, 1.15)
	end
	rescale()
	workspace.CurrentCamera:GetPropertyChangedSignal("ViewportSize"):Connect(rescale)

	buildHud()
	buildShop()
	buildDeposit()
	buildDialogue()
	buildToasts()
	timerLoop()

	lightTweenValue = Instance.new("NumberValue")
	lightTweenValue.Changed:Connect(function(value)
		local shown = math.floor(value + 0.5)
		if shown ~= displayedLight then
			displayedLight = shown
			hud.lightLabel.Text = formatNumber(shown)
		end
	end)
end

local function resolve(path)
	local node = game
	for part in string.gmatch(path, "[^%.]+") do
		if node == game then
			node = game:GetService(part)
		else
			node = node:FindFirstChild(part) or node:WaitForChild(part, 5)
		end
	end
	return node
end

-- сохраняем ID доната из прошлой установки
local preserved = {}
for _, file in ipairs(FILES) do
	if file.preserve then
		local root = resolve(file.service)
		local old = root:FindFirstChild("LastLighthouse") and root.LastLighthouse:FindFirstChild(file.name)
		if old and old:IsA("LuaSourceContainer") then
			preserved[file.name] = old.Source
		end
	end
end

	if old then

	end
end

	local root = resolve(file.service)
	local folder = root:FindFirstChild("LastLighthouse")

		folder = Instance.new("Folder")

		folder.Parent = root
	end

	script.Name = file.name
	script.Source = preserved[file.name] or file.source
	script.Parent = folder
	created = created + 1
end
